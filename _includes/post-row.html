{%- comment -%} One blog post as a row of an entry list (same markup as entry-row.html). Usage: {% include post-row.html post=post %} The Italian translation, if any, is the post with the same URL plus "-ita". {%- endcomment -%} {%- assign post = include.post -%} {%- assign category = post.categories | first -%} {%- assign kind = site.data.post_types | where: "key", category | first -%} {%- assign ita_url = post.url | replace: ".html", "-ita.html" -%} {%- assign ita = site.posts | where: "url", ita_url | first -%} {{ post.date | date: "%b" }} {{ post.date | date: "%Y" }} {% if kind %}{% endif %} {{ post.title }} {%- if ita %} Versione italiana{% endif -%}
